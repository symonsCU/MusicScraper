from playwright.sync_api import sync_playwright

from config import TOP100_URL


class BeatportScraper:

    def __init__(self):
        self.url = TOP100_URL

    def get_tracks(self):

        with sync_playwright() as p:

            browser = p.chromium.launch(
                headless=False
            )

            page = browser.new_page()

            page.goto(
                self.url,
                wait_until="networkidle",
                timeout=60000
            )

            # Give the page a few extra seconds to finish rendering
            page.wait_for_timeout(5000)

            print("\n===== PAGE TITLE =====")
            print(page.title())

            print("\n===== FIRST 30 DIVS WITH TEXT =====")

            divs = page.locator("div")
            count = divs.count()

            shown = 0

            for i in range(count):
                try:
                    text = divs.nth(i).inner_text().strip()

                    # Ignore empty divs
                    if len(text) == 0:
                        continue

                    print(f"\nDIV #{i}")
                    print("----------------------------")
                    print(text)

                    shown += 1

                    if shown >= 30:
                        break

                except Exception:
                    # Some elements can't be read; skip them
                    continue

            input("\nPress ENTER to close the browser...")

            browser.close()

        # We'll return real tracks later
        return []

from playwright.sync_api import sync_playwright
from config import TOP100_URL


class BeatportScraper:

    def __init__(self):
        self.url = TOP100_URL

    def get_tracks(self):

        with sync_playwright() as p:

            browser = p.chromium.launch(
                headless=False
            )

            page = browser.new_page()

            page.goto(
                self.url,
                wait_until="networkidle"
            )

            page.wait_for_timeout(5000)

            # Save a nicely formatted HTML file
            html = page.evaluate("""
                () => document.documentElement.outerHTML
            """)

            with open("beatport_live.html", "w", encoding="utf-8") as f:
                f.write(html)

            print("Saved beatport_live.html")

            browser.close()

        return []
